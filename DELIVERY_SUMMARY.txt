🛠️ UTILITIES REFERENCE
═══════════════════════════════════════════════════════════════════════════════

SAVE BOOKINGS
  saveFlightBooking(flightData)   // Save flight to localStorage
  saveHotelBooking(hotelData)     // Save hotel to localStorage

GET BOOKINGS
  getFlightBooking()              // Get current flight or null
  getHotelBooking()               // Get current hotel or null

CLEAR BOOKINGS
  clearFlightBooking()            // Remove flight from storage
  clearHotelBooking()             // Remove hotel from storage
  clearAllBookings()              // Remove all bookings

USAGE EXAMPLE
  import { saveFlightBooking, getFlightBooking } from '@/lib/bookingUtils';
  
  // Save
  saveFlightBooking({
    id: 'FL-' + Date.now(),
    from: 'JFK',
    to: 'LHR',
    departureDate: '2026-03-15',
    airline: 'Emirates',
    flightNumber: 'EK205',
    cabinClass: 'Business',
    price: 1299
  });
  
  // Retrieve
  const flight = getFlightBooking();
  console.log(flight?.price); // 1299
